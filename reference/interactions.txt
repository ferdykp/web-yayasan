    (function initPrestasiFiltering() {
      const levelButtons = document.querySelectorAll('[data-filter-type="level"]');
      const categoryButtons = document.querySelectorAll('[data-filter-type="category"]');
      const yearButtons = document.querySelectorAll('.year-btn');
      const cards = document.querySelectorAll('.prestasi-card');
      const recordCountEl = document.getElementById('record-count');

      let currentLevel = 'all';
      let currentCategory = 'all';

      function updateActiveStyles(buttonGroup, activeBtn) {
        buttonGroup.forEach(btn => {
          btn.classList.remove('bg-primary', 'text-on-primary', 'active-pill');
          btn.classList.add('bg-surface-container-high', 'text-on-surface-variant');
        });
        activeBtn.classList.remove('bg-surface-container-high', 'text-on-surface-variant');
        activeBtn.classList.add('bg-primary', 'text-on-primary', 'active-pill');
      }

      function filterCards() {
        let visibleCount = 0;
        cards.forEach(card => {
          const cardLevel = card.getAttribute('data-level');
          const cardCategory = card.getAttribute('data-category');

          const matchLevel = (currentLevel === 'all' || cardLevel === currentLevel);
          const matchCategory = (currentCategory === 'all' || cardCategory === currentCategory);

          if (matchLevel && matchCategory) {
            card.classList.remove('hidden');
            visibleCount++;
          } else {
            card.classList.add('hidden');
          }
        });

        if (recordCountEl) {
          recordCountEl.textContent = visibleCount;
        }
      }

      levelButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          currentLevel = btn.getAttribute('data-value');
          updateActiveStyles(levelButtons, btn);
          filterCards();
        });
      });

      categoryButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          currentCategory = btn.getAttribute('data-value');
          updateActiveStyles(categoryButtons, btn);
          filterCards();
        });
      });

      yearButtons.forEach(btn => {
        btn.addEventListener('click', () => {
          yearButtons.forEach(y => {
            y.classList.remove('bg-primary', 'text-on-primary', 'font-bold');
            y.classList.add('bg-surface-container-high', 'text-on-surface-variant');
          });
          btn.classList.remove('bg-surface-container-high', 'text-on-surface-variant');
          btn.classList.add('bg-primary', 'text-on-primary', 'font-bold');
        });
      });
    })();

  (function() {
    const categoryButtons = document.querySelectorAll('.category-btn');
    const searchInput = document.getElementById('article-search-input');
    const articles = document.querySelectorAll('.article-card');
    const emptyState = document.getElementById('no-articles-found');
    const countLabel = document.getElementById('article-count-label');
    const newsletterForm = document.getElementById('newsletter-form');
    const newsletterSuccess = document.getElementById('newsletter-success');

    let currentCategory = 'all';
    let searchQuery = '';

    function filterArticles() {
      let visibleCount = 0;

      articles.forEach(card => {
        const category = card.getAttribute('data-category');
        const textContent = card.innerText.toLowerCase();

        const matchesCategory = (currentCategory === 'all' || category === currentCategory);
        const matchesSearch = searchQuery === '' || textContent.includes(searchQuery.toLowerCase());

        if (matchesCategory && matchesSearch) {
          card.classList.remove('hidden');
          visibleCount++;
        } else {
          card.classList.add('hidden');
        }
      });

      if (countLabel) {
        countLabel.textContent = `Menampilkan ${visibleCount} Cerita`;
      }

      if (emptyState) {
        if (visibleCount === 0) {
          emptyState.classList.remove('hidden');
        } else {
          emptyState.classList.add('hidden');
        }
      }
    }

    categoryButtons.forEach(button => {
      button.addEventListener('click', () => {
        categoryButtons.forEach(btn => {
          btn.classList.remove('bg-primary-container', 'text-on-primary', 'shadow-sm');
          btn.classList.add('text-secondary');
        });

        button.classList.remove('text-secondary');
        button.classList.add('bg-primary-container', 'text-on-primary', 'shadow-sm');

        currentCategory = button.getAttribute('data-category');
        filterArticles();
      });
    });

    if (searchInput) {
      searchInput.addEventListener('input', (e) => {
        searchQuery = e.target.value.trim();
        filterArticles();
      });
    }

    if (newsletterForm) {
      newsletterForm.addEventListener('submit', (e) => {
        e.preventDefault();
        const input = document.getElementById('newsletter-email-input');
        if (input && input.value) {
          newsletterForm.classList.add('opacity-50', 'pointer-events-none');
          setTimeout(() => {
            newsletterForm.classList.add('hidden');
            if (newsletterSuccess) {
              newsletterSuccess.classList.remove('hidden');
              newsletterSuccess.classList.add('flex');
            }
          }, 400);
        }
      });
    }
  })();

    (function() {
      // 1. Reading Progress Bar
      window.addEventListener('scroll', function() {
        const winScroll = document.body.scrollTop || document.documentElement.scrollTop;
        const height = document.documentElement.scrollHeight - document.documentElement.clientHeight;
        const scrolled = (winScroll / height) * 100;
        const bar = document.getElementById('scroll-indicator');
        if (bar) {
          bar.style.width = scrolled + '%';
        }
      });

      // 2. Font Resizer
      let currentFontScale = 1;
      const articleContainer = document.getElementById('article-content-container');
      const btnIncrease = document.getElementById('btn-font-increase');
      const btnDecrease = document.getElementById('btn-font-decrease');

      if (btnIncrease && articleContainer) {
        btnIncrease.addEventListener('click', function() {
          if (currentFontScale < 1.25) {
            currentFontScale += 0.08;
            articleContainer.style.fontSize = (currentFontScale * 100) + '%';
          }
        });
      }

      if (btnDecrease && articleContainer) {
        btnDecrease.addEventListener('click', function() {
          if (currentFontScale > 0.85) {
            currentFontScale -= 0.08;
            articleContainer.style.fontSize = (currentFontScale * 100) + '%';
          }
        });
      }

      // 3. Like Appreciation Button
      const btnLike = document.getElementById('btn-like');
      const likeCounter = document.getElementById('like-counter');
      const likeIcon = document.getElementById('like-icon');
      let isLiked = false;

      if (btnLike && likeCounter && likeIcon) {
        btnLike.addEventListener('click', function() {
          let count = parseInt(likeCounter.innerText);
          if (!isLiked) {
            count += 1;
            likeCounter.innerText = count;
            likeIcon.style.fontVariationSettings = "'FILL' 1";
            btnLike.classList.add('bg-primary', 'text-on-primary');
            btnLike.classList.remove('bg-surface-container-lowest', 'text-primary');
            isLiked = true;
          } else {
            count -= 1;
            likeCounter.innerText = count;
            likeIcon.style.fontVariationSettings = "'FILL' 0";
            btnLike.classList.remove('bg-primary', 'text-on-primary');
            btnLike.classList.add('bg-surface-container-lowest', 'text-primary');
            isLiked = false;
          }
        });
      }

      // 4. Bookmark Toggle
      const btnBookmark = document.getElementById('btn-bookmark');
      const bookmarkIcon = document.getElementById('bookmark-icon');
      let isBookmarked = false;

      if (btnBookmark && bookmarkIcon) {
        btnBookmark.addEventListener('click', function() {
          isBookmarked = !isBookmarked;
          if (isBookmarked) {
            bookmarkIcon.style.fontVariationSettings = "'FILL' 1";
            alert('Artikel disimpan ke daftar bacaan Anda.');
          } else {
            bookmarkIcon.style.fontVariationSettings = "'FILL' 0";
          }
        });
      }
    })();

      const container = document.getElementById('modal-content-placeholder');
      container.innerHTML = `
        <div class="flex items-center gap-space-xs text-secondary mb-space-xs">
          <span class="material-symbols-outlined text-[20px]">apartment</span>
          <span class="font-label-md text-label-md uppercase tracking-wider">${data.location}</span>
        </div>
        <h3 class="font-headline-lg text-headline-lg text-primary font-normal mb-space-md">${data.title}</h3>
        
        <div class="mb-space-md">
          <h4 class="font-label-lg text-label-lg font-bold text-primary mb-space-xs uppercase tracking-wide">Rincian Spesifikasi &amp; Fasilitas:</h4>
          <ul class="space-y-space-xs font-body-md text-body-md text-on-surface-variant list-disc pl-5">
            ${data.specs.map(s => `<li>${s}</li>`).join('')}
          </ul>
        </div>

        <div class="p-space-md bg-surface-container rounded mb-space-md">
          <span class="font-label-sm text-label-sm font-bold text-primary block uppercase">Jadwal Operasional:</span>
          <p class="font-body-md text-body-md text-on-surface">${data.hours}</p>
        </div>

        <div class="flex justify-end gap-space-sm pt-space-xs">
          <a href="#reservasi-tur" onclick="closeFacilityModal()" class="px-space-lg py-space-xs rounded-full bg-primary text-on-primary font-label-md text-label-md font-bold hover:bg-primary-container transition-colors">
            Tinjau dalam Tur Kampus
          </a>
        </div>
      `;

      document.getElementById('facility-spec-modal').classList.remove('hidden');
    }
